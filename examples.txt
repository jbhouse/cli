function idk() {
  while getopts 'abf:v' flag; do
    case "${flag}" in
      a) a_flag='true' ;;
      b) b_flag='true' ;;
      f) files="${OPTARG}" ;;
      v) verbose='true' ;;
      *) print_usage
        exit 1 ;;
    esac
  done
}
